'use client';
import { useState } from 'react';

export default function ContactForm() {
  const [form, setForm] = useState({ name: '', email: '', message: '' });
  const [status, setStatus] = useState('');

  const handleSubmit = async (e) => {
    e.preventDefault();
    setStatus('Submitting to Cloud NoSQL...');
    try {
      const response = await fetch('http://localhost:5000/api/contact', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(form)
      });
      const data = await response.json();
      if (data.success) {
        setStatus('Message successfully recorded!');
        setForm({ name: '', email: '', message: '' });
      }
    } catch (err) {
      setStatus('Submission pipeline failure.');
    }
  };

  return (
    <form onSubmit={handleSubmit} className="p-6 bg-gray-100 rounded-lg max-w-md mx-auto space-y-4">
      <h2 className="text-xl font-bold">Contact Administration</h2>
      <input 
        type="text" placeholder="Your Name" required
        value={form.name} onChange={(e) => setForm({...form, name: e.target.value})}
        className="w-full p-2 border rounded text-black"
      />
      <input 
        type="email" placeholder="Your Email" required
        value={form.email} onChange={(e) => setForm({...form, email: e.target.value})}
        className="w-full p-2 border rounded text-black"
      />
      <textarea 
        placeholder="Type message here..." required
        value={form.message} onChange={(e) => setForm({...form, message: e.target.value})}
        className="w-full p-2 border rounded text-black h-24"
      />
      <button type="submit" className="bg-blue-600 text-white px-4 py-2 rounded shadow w-full">Submit Message</button>
      {status && <p className="text-sm font-medium mt-2 text-center text-blue-700">{status}</p>}
    </form>
  );
}
