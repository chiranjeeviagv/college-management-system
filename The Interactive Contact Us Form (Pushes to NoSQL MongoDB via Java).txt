import { useState } from 'react';

export default function ContactUs() {
  const [formData, setFormData] = useState({ name: '', email: '', subject: '', message: '' });
  const [status, setStatus] = useState({ type: '', text: '' });

  const handleChange = (e) => {
    setFormData({ ...formData, [e.target.name]: e.target.value });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setStatus({ type: 'info', text: 'Sending secure message...' });

    try {
      const response = await fetch(`${API_BASE_URL}/contact`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(formData),
      });

      if (response.ok) {
        setStatus({ type: 'success', text: 'Message successfully recorded in MongoDB Atlas!' });
        setFormData({ name: '', email: '', subject: '', message: '' });
      } else {
        setStatus({ type: 'error', text: 'Backend submission error. Please try again.' });
      }
    } catch (error) {
      setStatus({ type: 'error', text: 'Could not connect to Java API pipeline.' });
    }
  };

  return (
    <section id="contact-us" className="py-16 px-6 max-w-3xl mx-auto">
      <h2 className="text-3xl font-bold text-slate-900 mb-4 text-center">Contact Admissions Office</h2>
      <p className="text-slate-600 text-center mb-8">Have inquiries? Your submissions are securely evaluated and logged live using our NoSQL pipeline.</p>
      
      <form onSubmit={handleSubmit} className="space-y-4 bg-slate-50 p-8 rounded-xl border border-slate-200 shadow-sm">
        <div>
          <label className="block text-sm font-semibold text-slate-700 mb-1">Full Name</label>
          <input type="text" name="name" value={formData.name} onChange={handleChange} required className="w-full p-2 border border-slate-300 rounded focus:ring-2 focus:ring-blue-500 focus:outline-none" />
        </div>
        <div>
          <label className="block text-sm font-semibold text-slate-700 mb-1">Email Address</label>
          <input type="email" name="email" value={formData.email} onChange={handleChange} required className="w-full p-2 border border-slate-300 rounded focus:ring-2 focus:ring-blue-500 focus:outline-none" />
        </div>
        <div>
          <label className="block text-sm font-semibold text-slate-700 mb-1">Subject</label>
          <input type="text" name="subject" value={formData.subject} onChange={handleChange} required className="w-full p-2 border border-slate-300 rounded focus:ring-2 focus:ring-blue-500 focus:outline-none" />
        </div>
        <div>
          <label className="block text-sm font-semibold text-slate-700 mb-1">Your Message</label>
          <textarea name="message" rows="4" value={formData.message} onChange={handleChange} required className="w-full p-2 border border-slate-300 rounded focus:ring-2 focus:ring-blue-500 focus:outline-none"></textarea>
        </div>
        <button type="submit" className="w-full bg-blue-600 text-white font-bold py-2 px-4 rounded hover:bg-blue-700 transition">
          Submit Query to Cloud Pipeline
        </button>
        {status.text && (
          <p className={`mt-4 text-center font-semibold text-sm ${status.type === 'success' ? 'text-green-600' : status.type === 'error' ? 'text-red-600' : 'text-blue-500'}`}>
            {status.text}
          </p>
        )}
      </form>
    </section>
  );
}
