import { useEffect, useState } from 'react';

export default function DepartmentsAndPrograms() {
  const [departments, setDepartments] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch(`${API_BASE_URL}/departments`)
      .then((res) => res.json())
      .then((data) => {
        setDepartments(data);
        setLoading(false);
      })
      .catch((err) => console.error("Error loading academic data:", err));
  }, []);

  if (loading) return <div className="text-center p-8">Loading Academic Data...</div>;

  return (
    <section id="departments" className="py-16 px-6 max-w-7xl mx-auto">
      <h2 className="text-3xl font-bold text-slate-900 mb-8 border-b pb-2">Academic Departments & Programs</h2>
      <div className="grid md:grid-cols-2 gap-8">
        {departments.map((dept) => (
          <div key={dept.id} className="bg-white p-6 rounded-lg shadow-sm border border-slate-200">
            <h3 className="text-xl font-bold text-blue-600">{dept.name}</h3>
            <p className="text-sm text-slate-500 mb-4">HOD: {dept.headOfDepartment}</p>
            <h4 className="font-semibold text-slate-700 mb-2">Offered Programs:</h4>
            <ul className="list-disc pl-5 space-y-1 text-slate-600">
              {dept.programs?.map((prog) => (
                <li key={prog.id}>
                  <span className="font-medium text-slate-800">{prog.title}</span> — {prog.duration}
                </li>
              ))}
            </ul>
          </div>
        ))}
      </div>
    </section>
  );
}
